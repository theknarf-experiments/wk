# The wk QPA plugin, built OUT OF TREE against plugins/qt/sysroot.
#
# The obvious way to add a QPA plugin is in-tree, as
# qtbase/src/plugins/platforms/wk with a qt_internal_add_plugin() — that is
# what the vnc and linuxfb plugins do, and it is what a patch against qtbase
# would have to arrange. We do not, for two reasons:
#
#   * everything the plugin needs is already INSTALLED. Qt6FbSupportPrivate is
#     a findable package in the sysroot (fbconvenience is built unconditionally
#     for every platform — src/platformsupport/CMakeLists.txt has no guard on
#     it), and so are Qt6::GuiPrivate and Qt6::CorePrivate with the qpa/ and
#     private/ headers this plugin includes;
#   * an in-tree plugin means a qtbase patch, a qtbase reconfigure and a
#     qtbase rebuild for every edit to OUR code. patches/ is for changes to
#     upstream Qt. This is not upstream Qt.
#
# The cost of staying out of tree is that Qt's plugin CMake glue does not
# generate the Q_IMPORT_PLUGIN for us, so an app must write it by hand. See
# main.cpp.
cmake_minimum_required(VERSION 3.21)
project(qwk LANGUAGES C CXX)

# Qt6::CorePrivate and Qt6::GuiPrivate are TARGETS inside the Core and Gui
# packages, not packages of their own — asking for them as components fails
# with "Expected Config file at .../Qt6CorePrivateConfig.cmake does NOT exist".
# They carry qpa/*.h, qfreetypefontdatabase_p.h and qtimerinfo_unix_p.h.
# FbSupportPrivate genuinely IS its own package: it is an INTERNAL_MODULE, and
# it is the guest-side compositor (QFbScreen/QFbWindow/QFbBackingStore).
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui FbSupportPrivate)

set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# gfx-compat: the C shim over wasi:surface/graphics-context/frame-buffer, plus
# the wit-bindgen output regenerated by build-qpa.sh into gfx-compat/gen.
# wkgfx.c and gen/wkgfx.c go INTO this library; gen/wkgfx_component_type.o
# does not, because a static archive member nothing references is dropped by
# the linker and that object exists purely for its custom section. The app
# links it directly — see ../build-smoke.sh.
set(WK_GFX_COMPAT "${CMAKE_CURRENT_SOURCE_DIR}/../../gfx-compat" CACHE PATH
    "plugins/gfx-compat, the wasi-gfx C shim")

add_library(qwk STATIC
    main.cpp
    qwkbackingstore.cpp
    qwkcompositor.cpp
    qwkeventdispatcher.cpp
    qwkfontdatabase.cpp
    qwkinput.cpp
    qwkintegration.cpp
    qwkkeytranslator.cpp
    qwkscreen.cpp
    qwktheme.cpp
    qwkwindow.cpp
    "${WK_GFX_COMPAT}/wkgfx.c"
    "${WK_GFX_COMPAT}/gen/wkgfx.c"
)

target_include_directories(qwk PUBLIC "${WK_GFX_COMPAT}")

# QT_STATICPLUGIN is what makes Q_PLUGIN_METADATA emit
# qt_static_plugin_QWkIntegrationPlugin() instead of the dlopen entry points.
# Qt's own build sets it via qt_internal_add_plugin; out of tree it is ours.
target_compile_definitions(qwk PUBLIC QT_STATICPLUGIN)
target_compile_definitions(qwk PRIVATE QT_NO_FOREACH)

target_link_libraries(qwk PUBLIC
    Qt6::Core Qt6::CorePrivate
    Qt6::Gui  Qt6::GuiPrivate
    Qt6::FbSupportPrivate
)

install(TARGETS qwk ARCHIVE DESTINATION lib)
